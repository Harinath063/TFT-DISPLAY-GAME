#include <Adafruit_GFX.h>
#include <Adafruit_ILI9341.h>
#include <SPI.h>

#define TFT_CS   5
#define TFT_DC   2
#define TFT_RST  4

#define JOY_X 34
#define JOY_Y 35

Adafruit_ILI9341 tft = Adafruit_ILI9341(TFT_CS, TFT_DC, TFT_RST);

#define SCREEN_WIDTH 320
#define SCREEN_HEIGHT 240
#define BLOCK 10

int snakeX[100];
int snakeY[100];
int snakeLength = 3;

int dirX = 1;
int dirY = 0;

int foodX;
int foodY;

void spawnFood() {
  foodX = random(0, SCREEN_WIDTH / BLOCK) * BLOCK;
  foodY = random(0, SCREEN_HEIGHT / BLOCK) * BLOCK;
}

void drawFood() {
  tft.fillRect(foodX, foodY, BLOCK, BLOCK, ILI9341_RED);
}

void drawSnake() {
  for(int i=0;i<snakeLength;i++){
    tft.fillRect(snakeX[i], snakeY[i], BLOCK, BLOCK, ILI9341_GREEN);
  }
}

void eraseTail() {
  tft.fillRect(snakeX[snakeLength-1], snakeY[snakeLength-1], BLOCK, BLOCK, ILI9341_BLACK);
}

void moveSnake() {

  eraseTail();

  for(int i=snakeLength-1;i>0;i--){
    snakeX[i] = snakeX[i-1];
    snakeY[i] = snakeY[i-1];
  }

  snakeX[0] += dirX * BLOCK;
  snakeY[0] += dirY * BLOCK;
}

void checkJoystick() {

  int xVal = analogRead(JOY_X);
  int yVal = analogRead(JOY_Y);

  if(xVal < 1500){ dirX = -1; dirY = 0; }
  if(xVal > 3000){ dirX = 1; dirY = 0; }
  if(yVal < 1500){ dirX = 0; dirY = -1; }
  if(yVal > 3000){ dirX = 0; dirY = 1; }
}

void checkFood() {

  if(snakeX[0] == foodX && snakeY[0] == foodY){

    snakeLength++;
    spawnFood();
  }
}

void checkCollision() {

  if(snakeX[0] < 0 || snakeX[0] >= SCREEN_WIDTH || snakeY[0] < 0 || snakeY[0] >= SCREEN_HEIGHT){

    tft.fillScreen(ILI9341_RED);
    tft.setCursor(90,120);
    tft.setTextSize(3);
    tft.setTextColor(ILI9341_WHITE);
    tft.println("GAME OVER");

    while(1);
  }

  for(int i=1;i<snakeLength;i++){
    if(snakeX[0] == snakeX[i] && snakeY[0] == snakeY[i]){

      tft.fillScreen(ILI9341_RED);
      tft.setCursor(90,120);
      tft.setTextSize(3);
      tft.setTextColor(ILI9341_WHITE);
      tft.println("GAME OVER");

      while(1);
    }
  }
}

void setup() {

  tft.begin();
  tft.setRotation(1);
  tft.fillScreen(ILI9341_BLACK);

  for(int i=0;i<snakeLength;i++){
    snakeX[i] = 100 - i*BLOCK;
    snakeY[i] = 100;
  }

  spawnFood();
}

void loop() {

  checkJoystick();

  moveSnake();

  checkFood();

  checkCollision();

  drawFood();

  drawSnake();

  delay(120);
}
